// ------------------------------------------------------------------ setup
/** Run once from the editor (safe to re-run): creates the tabs, adds the channels, sets the 6-hour check, notes it in CAIOCC. */
function setupChannels() {
  ['Channels', 'ChannelStats', 'ChannelVideos', 'ChannelComments'].forEach(function (n) { ensureTab_(n, TABS[n]); });
  var have = {}; rows_('Channels').forEach(function (r) { have[r.key] = true; });
  CH_SEED.forEach(function (s) {
    if (have[s[0]]) return;
    append_('Channels', { key: s[0], platform: s[1], name: s[2], handle: s[3], channel_id: s[4], account: s[5], language_focus: s[6],
      goal_subs: s[7], goal_hours: s[8], goal_date: s[9], notes: s[10], url: chUrl_(s[1], s[3], s[4]) });
  });
  ScriptApp.getProjectTriggers().forEach(function (tr) { if (tr.getHandlerFunction() === 'channelsCheck') ScriptApp.deleteTrigger(tr); });
  ScriptApp.newTrigger('channelsCheck').timeBased().everyHours(6).create();
  var res = channelsCheck();
  try {
    var now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
    var reg = SpreadsheetApp.openById(CAIOCC_REGISTRY_ID), app = reg.getSheetByName('Applications'), head = app.getRange(1, 1, 1, app.getLastColumn()).getValues()[0];
    var v = app.getRange(2, 1, app.getLastRow() - 1, head.length).getValues();
    for (var i = 0; i < v.length; i++) if (String(v[i][head.indexOf('Name')]) === 'Crispus Profit Engine') {
      var old = String(v[i][head.indexOf('Notes')] || '');
      if (old.indexOf('Channels (v2.1)') < 0) {
        app.getRange(i + 2, head.indexOf('Notes') + 1).setValue(old + ' | Channels (v2.1, Nov 19 2026): ' + EXEC_URL + '?view=channels (PIN) — YouTube/Instagram/TikTok numbers, goals, comment reply queue; channelsCheck every 6h.');
        app.getRange(i + 2, head.indexOf('UpdatedAt') + 1).setValue(now);
      }
    }
  } catch (e) { Logger.log('CAIOCC note skipped: ' + e.message); }
  Logger.log('Channels ready. ' + res + ' | render ' + channelsPage_(token_(), '', '').getContent().length);
}
function chUrl_(platform, handle, id) {
  if (platform === 'YouTube') return id ? 'https://www.youtube.com/channel/' + id : (handle ? 'https://www.youtube.com/' + handle : '');
  if (!handle) return '';
  if (platform === 'Instagram') return 'https://www.instagram.com/' + handle.replace(/^@/, '') + '/';
  if (platform === 'TikTok') return 'https://www.tiktok.com/@' + handle.replace(/^@/, '');
  if (platform === 'Facebook') return 'https://www.facebook.com/' + handle.replace(/^@/, '');
  return '';
}

// ------------------------------------------------------------------ the check (trigger)
function channelsCheck() {
  if (typeof YouTube === 'undefined') return 'YouTube service is not turned on in this project.';
  ['Channels', 'ChannelStats', 'ChannelVideos', 'ChannelComments'].forEach(function (n) { ensureTab_(n, TABS[n]); });
  var out = [], today = today_();
  rows_('Channels').filter(function (c) { return c.platform === 'YouTube'; }).forEach(function (c) {
    try {
      var id = String(c.channel_id || '');
      if (!id && c.handle) {
        var f = YouTube.Channels.list('id', { forHandle: String(c.handle).replace(/^@/, '') });
        if (f.items && f.items.length) { id = f.items[0].id; update_('Channels', c._row, { channel_id: id, url: chUrl_('YouTube', c.handle, id) }); }
      }
      if (!id) { out.push(c.key + ': no channel id'); return; }
      var ch = YouTube.Channels.list('statistics,contentDetails', { id: id }).items;
      if (!ch || !ch.length) { out.push(c.key + ': not found'); return; }
      var st = ch[0].statistics, upl = ch[0].contentDetails.relatedPlaylists.uploads;
      update_('Channels', c._row, { subs: Number(st.subscriberCount || 0), views: Number(st.viewCount || 0), videos: Number(st.videoCount || 0), updated: new Date() });
      chSnapshot_(today, c.key, 'YouTube', Number(st.subscriberCount || 0), Number(st.viewCount || 0), Number(st.videoCount || 0), c.watch_hours, 'YouTube API');
      var vids = chVideos_(c.key, upl);
      var n = chComments_(c, id, vids);
      out.push(c.key + ': ' + st.subscriberCount + ' subs, ' + vids.length + ' videos, ' + n + ' new comments');
    } catch (e) { out.push(c.key + ': ' + e.message); }
  });
  props_().setProperty('CH_LAST_RUN', JSON.stringify({ at: new Date().toISOString(), result: out.join(' · ') }));
  return out.join(' · ');
}
function chSnapshot_(date, key, platform, subs, views, videos, hours, source) {
  var r = rows_('ChannelStats').filter(function (x) { return x.date === date && x.key === key; })[0];
  var o = { date: date, key: key, platform: platform, subs: subs, views: views, videos: videos, watch_hours: hours === undefined ? '' : hours, source: source };
  if (r) update_('ChannelStats', r._row, o); else append_('ChannelStats', o);
}
function chVideos_(key, uploads) {
  var ids = [], tok = '';
  do {
    var pl = YouTube.PlaylistItems.list('contentDetails', { playlistId: uploads, maxResults: 50, pageToken: tok });
    (pl.items || []).forEach(function (it) { ids.push(it.contentDetails.videoId); });
    tok = pl.nextPageToken || '';
  } while (tok && ids.length < 500);
  var existing = {}; rows_('ChannelVideos').forEach(function (r) { existing[r.video_id] = r._row; });
  var vids = [];
  for (var i = 0; i < ids.length; i += 50) {
    var res = YouTube.Videos.list('snippet,statistics', { id: ids.slice(i, i + 50).join(',') });
    (res.items || []).forEach(function (v) {
      var lang = String(v.snippet.defaultAudioLanguage || v.snippet.defaultLanguage || 'en').slice(0, 2).toLowerCase();
      var o = { key: key, video_id: v.id, title: v.snippet.title, published: String(v.snippet.publishedAt).slice(0, 10), lang: lang,
        views: Number(v.statistics.viewCount || 0), likes: Number(v.statistics.likeCount || 0), comments: Number(v.statistics.commentCount || 0), updated: today_() };
      if (existing[v.id]) update_('ChannelVideos', existing[v.id], o); else append_('ChannelVideos', o);
      vids.push(o);
    });
  }
  return vids;
}
function chComments_(c, channelId, vids) {
  var byId = {}; vids.forEach(function (v) { byId[v.video_id] = v; });
  var known = {}; rows_('ChannelComments').forEach(function (r) { known[r.comment_id] = r; });
  var added = 0, tok = '', pages = 0;
  do {
    var res;
    try { res = YouTube.CommentThreads.list('snippet,replies', { allThreadsRelatedToChannelId: channelId, maxResults: 100, pageToken: tok, textFormat: 'plainText' }); }
    catch (e) { if (/disabled/i.test(e.message)) return 0; throw e; }
    (res.items || []).forEach(function (th) {
      var top = th.snippet.topLevelComment.snippet, cid = th.snippet.topLevelComment.id;
      var mine = top.authorChannelId && top.authorChannelId.value === channelId;
      var answered = (th.replies && th.replies.comments || []).some(function (r) { return r.snippet.authorChannelId && r.snippet.authorChannelId.value === channelId; });
      var k = known[cid];
      if (mine) return;
      if (k) { if (answered && k.status === 'needs reply') update_('ChannelComments', k._row, { status: 'replied', updated: today_() }); return; }
      var v = byId[th.snippet.videoId] || { title: '', lang: 'en' };
      var cat = chCategory_(top.textDisplay);
      var o = { comment_id: cid, key: c.key, video_id: th.snippet.videoId, video_title: v.title, lang: v.lang, author: top.authorDisplayName, text: String(top.textDisplay).slice(0, 2000),
        published: String(top.publishedAt).slice(0, 10), category: cat, status: answered ? 'replied' : (cat === 'spam' ? 'skip' : 'needs reply'),
        draft: answered || cat === 'spam' ? '' : chDraft_(top.textDisplay, v.lang, cat, v.title, c.name), updated: today_() };
      append_('ChannelComments', o); added++;
    });
    tok = res.nextPageToken || ''; pages++;
  } while (tok && pages < 5);
  return added;
}
function chCategory_(text) {
  var t = String(text || '');
  if (CH_CRISIS.test(t)) return 'crisis';
  if (CH_SPAM.test(t)) return 'spam';
  if (CH_PRAYER.test(t)) return 'prayer';
  if (/\?|¿|？/.test(t)) return 'question';
  if (CH_THANKS.test(t)) return 'thanks';
  return 'other';
}
/** Drafts a reply in the video's language. Crisis comments always use the fixed helpline text (never AI). */
function chDraft_(text, lang, cat, title, channel) {
  var L = CH_TEMPLATES.crisis[lang] ? lang : 'en';
  if (cat === 'crisis') return CH_TEMPLATES.crisis[L];
  var tpl = (CH_TEMPLATES[cat] || CH_TEMPLATES.other)[L] || CH_TEMPLATES.other[L];
  var sys = 'You reply to YouTube comments for the Christian channel "' + channel + '". Write ONE reply of 1 to 3 short sentences in ' + (CH_LANG_NAME[L] || 'the same language as the comment') +
    ' (if the comment is clearly in another language, use the comment\'s language). Warm, plain, respectful; no flattery, no emojis except at most one 🙏. ' +
    'If it is a prayer request: acknowledge it, say the team is praying, and add one short fitting Bible verse with its reference. If it is a question about the app: the app is https://covenant.crispusmundo.com and is in English for now; never call it free. ' +
    'Write "Lord" in normal case, never all capitals. Do not give medical, legal or financial advice. Output only the reply text.';
  try {
    var r = UrlFetchApp.fetch('https://text.pollinations.ai/openai', { method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      payload: JSON.stringify({ model: 'openai', messages: [{ role: 'system', content: sys }, { role: 'user', content: 'Video: ' + title + '\nComment: ' + String(text).slice(0, 1500) }] }) });
    if (r.getResponseCode() === 200) {
      var body = r.getContentText(), out = body;
      try { var j = JSON.parse(body); out = j.choices && j.choices[0] && j.choices[0].message ? j.choices[0].message.content : body; } catch (e) { }
      out = String(out || '').trim().replace(/^["“]|["”]$/g, '').replace(/\bLORD\b/g, 'Lord');
      if (out && out.length < 700 && !/\bfree\b|gratuit|gratis|grátis|免费|मुफ़्त|bure/i.test(out)) return out;
    }
  } catch (e) { /* fall back to the template */ }
  return tpl;
}

// ------------------------------------------------------------------ actions
function chAct_(p) {
  if (p.a === 'chstatus') {
    var r = rows_('ChannelComments').filter(function (x) { return x.comment_id === p.id; })[0];
    if (!r) throw new Error('Comment not found');
    if (CH_STATUSES.indexOf(p.status) < 0) throw new Error('Unknown status');
    update_('ChannelComments', r._row, { status: p.status, updated: today_() }); return 'Marked ' + p.status + '.';
  }
  if (p.a === 'chdraft') {
    var d = rows_('ChannelComments').filter(function (x) { return x.comment_id === p.id; })[0];
    if (!d) throw new Error('Comment not found');
    update_('ChannelComments', d._row, { draft: String(p.draft || '').slice(0, 2000), updated: today_() }); return 'Draft saved.';
  }
  if (p.a === 'chnums') {
    var c = rows_('Channels').filter(function (x) { return x.key === p.key; })[0];
    if (!c) throw new Error('Channel not found');
    var o = { updated: new Date() };
    ['subs', 'views', 'videos', 'watch_hours'].forEach(function (k) { if (p[k] !== undefined && p[k] !== '') o[k] = num_(p[k]); });
    if (p.handle !== undefined) { o.handle = String(p.handle).trim().slice(0, 80); o.url = chUrl_(c.platform, o.handle, c.channel_id); }
    ['goal_subs', 'goal_hours'].forEach(function (k) { if (p[k] !== undefined && p[k] !== '') o[k] = num_(p[k]); });
    update_('Channels', c._row, o);
    chSnapshot_(today_(), c.key, c.platform, o.subs !== undefined ? o.subs : c.subs, o.views !== undefined ? o.views : c.views, o.videos !== undefined ? o.videos : c.videos,
      o.watch_hours !== undefined ? o.watch_hours : c.watch_hours, 'typed in');
    return 'Saved ' + c.name + ' (' + c.platform + ').';
  }
  if (p.a === 'chrun') return channelsCheck() || 'Checked.';
  return '';
}
var _chOrigAct = act_;
act_ = function (p) { if (p && p.a && String(p.a).indexOf('ch') === 0) return chAct_(p); return _chOrigAct(p); };

function chSpark_(vals, color) {
  vals = vals.map(Number).filter(function (x) { return isFinite(x); });
  if (vals.length < 2) return '';
  var mx = Math.max.apply(null, vals), mn = Math.min.apply(null, vals), w = 120, h = 30, rng = mx - mn || 1;
  var pts = vals.map(function (v, i) { return (i * w / (vals.length - 1)).toFixed(1) + ',' + (h - 2 - (v - mn) / rng * (h - 4)).toFixed(1); }).join(' ');
  return '<svg width="' + w + '" height="' + h + '" viewBox="0 0 ' + w + ' ' + h + '" role="img" aria-label="trend"><polyline fill="none" stroke="' + color + '" stroke-width="2" points="' + pts + '"/></svg>';
}
function chBar_(val, goal) {
  if (!goal) return '';
  var pct = Math.max(0, Math.min(100, Math.round(num_(val) / num_(goal) * 100)));
  return '<div style="background:#eee;border-radius:6px;height:8px;margin:4px 0"><div style="width:' + pct + '%;background:#b8862b;height:8px;border-radius:6px"></div></div><div class="muted">' + pct + '% of ' + Math.round(num_(goal)).toLocaleString() + '</div>';
}
